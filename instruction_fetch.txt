`timescale 1ns / 1ps

module instruction_fetch (input clk,reset,stall, input [15:0] instruction,output reg [15:0] instruction_d1 );

 always @ (posedge clk) begin
  if (reset) 
   instruction_d1 <=0;
  else if (stall)
   instruction_d1 <= instruction_d1;
  else
   instruction_d1 <= instruction;
 end
endmodule
