`timescale 1ns / 1ps

module pipeline_alu #(parameter N=8)(
    input  [N-1:0]     A,
    input  [N-1:0]     B,

    input              CLK,
    input              reset,

    input  [4:0]       opcode,

    output reg [N-1:0] Y,
    output reg [3:0]   flag
);

 
    reg [N-1:0] a;
    reg [N-1:0] b;
    reg [4:0] op;

    always @(posedge CLK) begin
        if (reset) begin
            a  <= {N{1'b0}};
            b  <= {N{1'b0}};
            op <= 5'b00000;

        end

        else begin

            a  <= A;
            b  <= B;
            op <= opcode;

        end

    end


    // STAGE 2 : Execute
   
    reg [N-1:0] result;

    reg [N:0] temp;

    reg c;
    reg v;


    always @(posedge CLK) begin

        if (reset) begin

            result <= {N{1'b0}};

            c <= 1'b0;
            v <= 1'b0;

        end

        else begin

            c <= 1'b0;
            v <= 1'b0;


            case (op)

                // ================================================
                // NOP
                // ================================================

                5'b00000: begin

                    result <= result;

                end


                // ================================================
                // ADD
                // ================================================

                5'b00001: begin

                    temp = {1'b0,a} + {1'b0,b};

                    result <= temp[N-1:0];

                    c <= temp[N];

                    v <= (a[N-1] == b[N-1]) &&
                         (temp[N-1] != a[N-1]);

                end


                // ================================================
                // SUB
                // ================================================

                5'b00010: begin

                    temp = {1'b0,a} - {1'b0,b};

                    result <= temp[N-1:0];

                    c <= temp[N];

                    v <= (a[N-1] != b[N-1]) &&
                         (temp[N-1] != a[N-1]);

                end


                // ================================================
                // INC
                // ================================================

                5'b00011: begin

                    temp = {1'b0,a} + 1'b1;

                    result <= temp[N-1:0];

                    c <= temp[N];

                    v <= (a[N-1] == 1'b0) &&
                         (temp[N-1] == 1'b1);

                end


                // ================================================
                // DEC
                // ================================================

                5'b00100: begin

                    temp = {1'b0,a} - 1'b1;

                    result <= temp[N-1:0];

                    c <= temp[N];

                    v <= (a[N-1] == 1'b1) &&
                         (temp[N-1] == 1'b0);

                end


                // ================================================
                // AND
                // ================================================

                5'b00101:
                    result <= a & b;


                // ================================================
                // OR
                // ================================================

                5'b00110:
                    result <= a | b;


                // ================================================
                // XOR
                // ================================================

                5'b00111:
                    result <= a ^ b;


                // ================================================
                // NOT
                // ================================================

                5'b01000:
                    result <= ~a;


                // ================================================
                // SHIFT LEFT LOGICAL
                // ================================================

                5'b01001:
                    result <= a << b;


                // ================================================
                // SHIFT RIGHT LOGICAL
                // ================================================

                5'b01010:
                    result <= a >> b;


                // ================================================
                // SHIFT RIGHT ARITHMETIC
                // ================================================

                5'b01011:
                    result <= $signed(a) >>> b;


                // ================================================
                // SET LESS THAN - SIGNED
                // ================================================

                5'b01100: begin

                    if ($signed(a) < $signed(b))
                        result <= {{(N-1){1'b0}},1'b1};
                    else
                        result <= {N{1'b0}};

                end


                // ================================================
                // SET LESS THAN - UNSIGNED
                // ================================================

                5'b01101: begin

                    if (a < b)
                        result <= {{(N-1){1'b0}},1'b1};
                    else
                        result <= {N{1'b0}};

                end


                // ================================================
                // ADDI
                // ================================================

                5'b01110: begin

                    temp = {1'b0,a} + {1'b0,b};

                    result <= temp[N-1:0];

                    c <= temp[N];

                    v <= (a[N-1] == b[N-1]) &&
                         (temp[N-1] != a[N-1]);

                end


                // ================================================
                // SUBI
                // ================================================

                5'b01111: begin

                    temp = {1'b0,a} - {1'b0,b};

                    result <= temp[N-1:0];

                    c <= temp[N];

                    v <= (a[N-1] != b[N-1]) &&
                         (temp[N-1] != a[N-1]);

                end


                // ================================================
                // ANDI
                // ================================================

                5'b10000:
                    result <= a & b;


                // ================================================
                // ORI
                // ================================================

                5'b10001:
                    result <= a | b;


                // ================================================
                // XORI
                // ================================================

                5'b10010:
                    result <= a ^ b;


                // ================================================
                // LDI
                //
                // B contains immediate
                // ================================================

                5'b10011:
                    result <= b;

                default:
                    result <= {N{1'b0}};

            endcase

        end

    end

    // STAGE 3 : Output / Flag

    always @(posedge CLK) begin

        if (reset) begin

            Y    <= {N{1'b0}};
            flag <= 4'b0000;

        end

        else begin

            Y <= result;

            // Carry
            flag[3] <= c;

            // Zero
            flag[2] <= (result == {N{1'b0}});

            // Negative
            flag[1] <= result[N-1];

            // Overflow
            flag[0] <= v;

        end

    end

endmodule
