`timescale 1ns / 1ps

module id_ex (
    input clk,
    input reset,
    input stall,

    input [4:0] opcode,

    input [2:0] rd,
    input [2:0] rs1,
    input [2:0] rs2,

    input [7:0] immediate,
    input       alu_src_imm,


    output reg [4:0] opcode_d2,

    output reg [2:0] rd_d2,
    output reg [2:0] rs1_d2,
    output reg [2:0] rs2_d2,

    output reg [7:0] immediate_d2,
    output reg       alu_src_imm_d2
);


always @(posedge clk) begin

    if (reset) begin

        opcode_d2      <= 5'b00000;

        rd_d2          <= 3'b000;
        rs1_d2         <= 3'b000;
        rs2_d2         <= 3'b000;

        immediate_d2   <= 8'b00000000;

        alu_src_imm_d2 <= 1'b0;

    end

    else if (stall) begin

        opcode_d2      <= opcode_d2;

        rd_d2          <= rd_d2;
        rs1_d2         <= rs1_d2;
        rs2_d2         <= rs2_d2;

        immediate_d2   <= immediate_d2;

        alu_src_imm_d2 <= alu_src_imm_d2;

    end

    else begin

        opcode_d2      <= opcode;

        rd_d2          <= rd;
        rs1_d2         <= rs1;
        rs2_d2         <= rs2;

        immediate_d2   <= immediate;

        alu_src_imm_d2 <= alu_src_imm;

    end

end

endmodule
